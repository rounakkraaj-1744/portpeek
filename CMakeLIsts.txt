cmake_minimum_required(VERSION 3.16)

project(portpeek
    VERSION 0.1.0
    LANGUAGES C
)

include(CTest)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_executable(portpeek
    src/main.c
    src/cli/parser.c
    src/cli/commands.c
)

target_include_directories(portpeek
    PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(portpeek PRIVATE
        -Wall
        -Wextra
        -Wpedantic
        -Werror
    )
endif()

if(BUILD_TESTING)
    add_executable(portpeek_cli_tests
        tests/unit/test_cli.c
        src/cli/parser.c
    )
    target_include_directories(portpeek_cli_tests PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/include
    )
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(portpeek_cli_tests PRIVATE
            -Wall
            -Wextra
            -Wpedantic
            -Werror
        )
    endif()
    add_test(NAME cli_parser COMMAND portpeek_cli_tests)
endif()
